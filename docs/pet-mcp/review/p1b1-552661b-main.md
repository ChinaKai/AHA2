## 1. 本批关闭的 4 行

agent 渠道 HTTP 一共 8 行 pending，本批关闭 4 行，剩下 4 行见第 5 节。账本 HTTP 关闭数从 53 升到 57（434 条），普通 Task 工具仍是 61/61。

| 操作 | 状态 | 宠物入口 |
|---|---|---|
| `POST /api/v1/agent/channel/replies` | direct | `aha2_channel.channel_reply` |
| `POST /api/v1/agent/channel/reply-decision` | equivalent | `aha2_channel.channel_reply_decision`（end）；`aha2_channel.channel_reply`（continue） |
| `GET /api/v1/agent/channel/contacts` | equivalent | `aha2_channel.channel_contacts` |
| `POST /api/v1/agent/channel/messages` | protected_handoff | `aha2_pet.aha_send_channel_message` |

- **replies**：HTTP handler 和 MCP 工具调用的是同一个 `channel.Service.SendAgentChannelReply`。工具用 Turn+receipt+正文生成稳定 request_id，所以重试是幂等的。宠物的 aha2_channel 只在 owner 激活的路由指向 task_pet 时注入。
- **reply-decision**：HTTP 接受 `continue|end`，工具只给 `end`。`continue` 只在机器人对话里影响投递：`channelGroupReplyStateTx` 对机器人发来的消息，只有 decision 为 continue 才放行回复投递；而 `EnqueueAgentChannelReplyWithMentions` 在回复机器人时会自己把空 decision 写成 continue。Agent API 文档也写明，带 receipt 的 replies 就是 continue 信号。`TestAgentChannelReplyIsReceiptBoundAndIdempotent`（store）断言了回复机器人后 decision=continue，所以标 equivalent。
- **contacts**：同一个服务 `AgentTaskChannelContacts`。工具返回身份链接 ID、显示名、角色、协作角色和是否机器人，不返回 provider 用户 ID。HTTP body 里另外附带的路由摘要（会话、实例、路由 ID）工具没有，所以标 equivalent，不标 direct。
- **messages（主动外发）**：普通 Task 由 Main 直接发阻塞协调消息。宠物没有直接外发入口，只能用 `aha_send_channel_message` 提交 `external_channel_send` 审批：宠物不能自批，owner 批准后才走同一个 `sendTaskChannelOutreach`。所以标 protected_handoff。

## 2. 发现并修复的缺陷（552661b）

宠物外发审批被 owner 批准后，执行必然失败，返回“审批已批准但执行失败：sql: no rows in result set”。

AND task_id=?`，合成 ID 查不到，整个事务回滚。

原有的 `TestPetChannelSendRequiresApprovalAndExecutes`（app）用的是假 sender，从来没有走到真实渠道服务，所以一直没被发现。

func (s *Service) PetRequestChannelSend(ctx context.Context, input PetChannelSendInput) (domain.PetApproval, error) {
	if _, err := s.petChannelWritable(ctx); err != nil {

	}
	reasonText, reasonErr := requirePetApprovalReason(input.Reason)
	if reasonErr != nil {
		return domain.PetApproval{}, reasonErr
	}
	taskID := strings.TrimSpace(input.TaskID)
	message := strings.TrimSpace(input.Message)
	if taskID == "" {
		return domain.PetApproval{}, fmt.Errorf("task_id 不能为空")
	}
	if message == "" || len([]rune(message)) > 1000 {
		return domain.PetApproval{}, fmt.Errorf("消息不能为空且不能超过 1000 个字符")
	}
	if len(input.MentionIdentityLinkIDs) > 5 {
		return domain.PetApproval{}, fmt.Errorf("最多提及 5 个渠道联系人")
	}
	task, err := s.store.Task(ctx, taskID)
	if err != nil || task.ReadOnly {
		return domain.PetApproval{}, fmt.Errorf("Task 不存在或只读")
	}
	route, err := s.store.ActiveChannelTaskRouteForTask(ctx, task.ID)
	if err != nil {
		return domain.PetApproval{}, fmt.Errorf("该 Task 没有活动渠道路由，请先绑定渠道")
	}
	ownerID, err := s.store.PrimaryOwnerID(ctx)
	if err != nil {
		return domain.PetApproval{}, fmt.Errorf("本机还没有 Owner 账号")
	}
	destination, err := s.store.ChannelDestination(ctx, ownerID, route.ConversationID)
	if err != nil {
		return domain.PetApproval{}, fmt.Errorf("渠道目的地不存在，可能已失效")
	}
	if destination.EndpointKind != domain.ChannelEndpointGroupDigitalHuman || destination.Status != "active" {
		return domain.PetApproval{}, fmt.Errorf("只有群聊数字人渠道支持对外发送")
	}
	contacts, err := s.store.TaskChannelContacts(ctx, task.ID, route.InstanceID, route.ConversationID, 50)
	if err != nil {
		return domain.PetApproval{}, err
	}
	available := map[string]bool{}
	for _, contact := range contacts {
		available[contact.IdentityLinkID] = true
	}
	mentions := make([]string, 0, len(input.MentionIdentityLinkIDs))
	seen := map[string]bool{}
	for _, raw := range input.MentionIdentityLinkIDs {
		identityID := strings.TrimSpace(raw)
		if identityID == "" || seen[identityID] || !available[identityID] {
			return domain.PetApproval{}, fmt.Errorf("渠道联系人 %s 不在当前群聊", identityID)

		mentions = append(mentions, identityID)
	}
	requestID := domain.NewID("pet_channel_send")

		Title:   "对外渠道发送：" + destination.DisplayName,

func (s *Server) resolvePetApproval(writer http.ResponseWriter, request *http.Request) {
	var input struct {
		Decision       string `json:"decision"`
		Note           string `json:"note"`
		DecisionReason string `json:"decision_reason"`
	}
	if err := decodeJSON(request, &input); err != nil {
		writeError(writer, http.StatusBadRequest, "invalid_json")
		return
	}
	reason := strings.TrimSpace(input.DecisionReason)
	if reason == "" {
		reason = input.Note
	}
	audited := false
	item, err := s.app.ResolvePetApprovalBeforeExecution(request.Context(), request.PathValue("id"), input.Decision, reason, func(item domain.PetApproval) {
		audited = true
		s.audit(request, "pet.approval.resolve", "pet_approval", item.ID, map[string]any{"status": item.Status})
	})
	if err != nil {
		status := http.StatusBadRequest
		if strings.Contains(err.Error(), "已经处理过") {
			status = http.StatusConflict
		}
		writeJSON(writer, status, map[string]any{"ok": false, "error": "pet_approval_resolve_failed", "message": err.Error()})
		return
	}
	if !audited {
		s.audit(request, "pet.approval.resolve", "pet_approval", item.ID, map[string]any{"status": item.Status})
	}
	// A heavy delete was claimed and is running in the background; report the
	// acceptance instead of holding this connection open for the transaction.
	if item.Status == "executing" {
		writeJSON(writer, http.StatusAccepted, map[string]any{"ok": true, "status": "executing"})
		return
	}
	writeJSON(writer, http.StatusOK, map[string]any{"ok": true, "approval": item})
}
