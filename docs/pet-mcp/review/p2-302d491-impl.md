## A. P2 实现现状（审查对象，非本轮 diff）

要点：
- 浏览器 16 个、桌面 5 个、Workspace 6 个工具都是 operation adapter：每个工具固定一个原 Agent handler、方法、路径字段和 schema，不接受任意 URL/方法/Task 身份，不携带 owner Cookie/CSRF；结果只透出白名单字段或有界字节，错误只透出固定代码。
- 注入与调用门禁：`browserMCPInjected` 要求当前 Task 有 owner 授予的有效浏览器 grant；桌面要求 owner 打开的共享；Workspace 要求绑定的 Workspace。宠物额外要求实时查看权限（`petDomainReadable`），写操作要求 Main、可写快照，宠物还要 Task 管理权限（`hardwareMCPSnapshotWritable`）。渠道 Turn 一律不注入这些域。
- 共享目标、profile、能力范围、登录、Cookie 值、桌面目标与模式都留在 owner Web；工具无法扩大。

func (s *Server) browserMCPCookieMetadata(writer http.ResponseWriter, request *http.Request) {
	service, grant, ok := s.agentBrowserGrantTarget(writer, request)
	if !ok {
		return
	}
	items, truncated, err := service.ExportCookies(request.Context(), browserActor(request), grant.ID, request.URL.Query().Get("url"), request.URL.Query().Get("domain"))
	if err != nil {
		writeBrowserFailure(writer, err)
		return
	}
	metadata := make([]map[string]any, 0, len(items))
	for _, item := range items {
		metadata = append(metadata, map[string]any{"name": item.Name, "domain": item.Domain, "path": item.Path, "expires": item.Expires, "http_only": item.HTTPOnly, "secure": item.Secure, "same_site": item.SameSite})
	}
	writeJSON(writer, http.StatusOK, map[string]any{"ok": true, "cookies": metadata, "truncated": truncated})
}

// Mutation results expose only the identifiers needed for the next operation.

func browserMCPSafeResult(name string, result map[string]any) map[string]any {
	view := map[string]any{"ok": result["ok"]}
	switch name {
	case "browser_tab_open", "browser_tab_bind_current", "browser_tab_activate", "browser_navigate":
		if tab, ok := result["tab"].(map[string]any); ok {
			url, _ := tab["url"].(string)
			safe, site := sanitizeBrowserMCPURL(url)
			view["tab"] = map[string]any{"id": tab["id"], "title": tab["title"], "url": safe, "site": site, "active": tab["active"], "focused": tab["focused"]}
		}
	case "browser_bookmarks_list":
		items := []map[string]any{}
		if bookmarks, ok := result["bookmarks"].([]any); ok {
			for _, raw := range bookmarks {
				if item, ok := raw.(map[string]any); ok {
					url, _ := item["url"].(string)
					safe, _ := sanitizeBrowserMCPURL(url)
					items = append(items, map[string]any{"title": item["title"], "url": safe, "folder": item["folder"]})
				}
			}
		}
		view["bookmarks"] = items
	case "browser_downloads_list":
		view["downloads"] = result["downloads"]
	case "browser_cookie_metadata":
		view["cookies"] = result["cookies"]
	case "browser_frame", "browser_download_get":
		return result
	}
	return view
}

func (s *Server) browserMCPToolsFor(write bool) []mcpTool {
	tools := agentBrowserMCPTools()
	for _, operation := range s.browserMCPOperations() {
		if !operation.Write || write {
			tools = append(tools, operation.Tool)
		}
	}
	return tools
}

// AgentBrowserMCPAllowed is the single server-side gate for the shared-browser

	return s.browserMCPInjected(ctx, call.Task, call.Turn.InboxBatchID)

// browserMCPInjected reports whether the shared-browser domain belongs to this
// Turn. Its trigger is the presence of an active browser grant for the Task:
// without a lease the tools could only report "no grant", so the domain is not
// advertised at all. Pet access requires its live view permission; channel-hosted
// Turns remain excluded. A grant whose session is already closed does not count, which
// mirrors the Agent-facing HTTP view.
func (s *Service) browserMCPInjected(ctx context.Context, task domain.Task, inboxBatchID string) bool {
	if !s.petDomainReadable(ctx, task) {
		return false
	}
	if s.agentMCPChannelBound(ctx, task.ID, inboxBatchID) {
		return false
	}
	grants, err := s.store.BrowserTaskGrants(ctx, task.ID)
	if err != nil {
		return false
	}
	now := s.now().UTC()
	for _, grant := range grants {
		if grant.Status(now) != domain.BrowserGrantStatusActive {
			continue
		}
		session, sessionErr := s.store.BrowserSession(ctx, grant.SessionID)
		if sessionErr == nil && session.State != domain.BrowserSessionStateClosed {
			return true
		}
	}
	return false
}

func runAgentMCPOperation(request *http.Request, op agentMCPOperation, raw json.RawMessage) (result map[string]any, err error) {
	if request.Context().Err() != nil {
		return nil, errors.New("operation cancelled")
	}
	decoder := json.NewDecoder(bytes.NewReader(raw))
	decoder.UseNumber()
	var arguments map[string]any
	if decoder.Decode(&arguments) != nil || validateMCPOperation(arguments, op.Tool.InputSchema) != nil {
		return nil, errors.New("invalid tool arguments")
	}
	if decoder.Decode(new(any)) != io.EOF {
		return nil, errors.New("invalid tool arguments")
	}
	child := request.Clone(context.WithValue(request.Context(), agentOperationContextKey{}, op.Tool.Name))
	child.Method = op.Method
	child.Header = make(http.Header)
	child.URL = &url.URL{Path: request.URL.Path}
	query := url.Values{}
	for field, pathName := range op.PathFields {
		if value, ok := arguments[field]; ok {
			child.SetPathValue(pathName, fmt.Sprint(value))
			delete(arguments, field)
		}
	}
	for _, field := range op.QueryFields {
		if value, ok := arguments[field]; ok {
			query.Set(field, fmt.Sprint(value))
			delete(arguments, field)
		}
	}
	child.URL.RawQuery = query.Encode()
	body, err := json.Marshal(arguments)
	if err != nil {
		return nil, errors.New("invalid tool arguments")
	}
	child.Header.Set("Content-Type", "application/json")
	child.Body = io.NopCloser(bytes.NewReader(body))
	child.ContentLength = int64(len(body))
	response := &mcpOperationResponse{header: make(http.Header)}
	defer func() {
		if recovered := recover(); recovered != nil {
			if recovered == http.ErrAbortHandler {
				result = nil
				err = errors.New("operation stream interrupted")
				return
			}
			panic(recovered)
		}
	}()
	op.Handler(response, child)
	// Raw runtime errors can contain endpoints or credentials. Only a fixed
	// classification leaves this boundary, and no body is logged here.
	if response.status >= 400 {
		return nil, &agentOperationError{Status: response.status, Code: safeAgentOperationCode(response.body.Bytes())}
	}
	if response.overflow {
		return nil, errors.New("operation response exceeds limit")
	}
	if op.Binary {
		return map[string]any{"media_type": response.header.Get("Content-Type"), "content_base64": base64.StdEncoding.EncodeToString(response.body.Bytes())}, nil
	}
	if json.Unmarshal(response.body.Bytes(), &result) != nil {
		return nil, errors.New("operation result unavailable")
	}
	if op.ResultFields != nil {
		view := map[string]any{}
		for _, key := range op.ResultFields {
			if value, ok := result[key]; ok {
				view[key] = value
			}
		}
		return view, nil
	}
	return result, nil
}

// Codes are copied only from this fixed vocabulary, never from an upstream
// message, response header or arbitrary provider error.
func safeAgentOperationCode(body []byte) string {
	var failure struct {
		Error string `json:"error"`

	}
